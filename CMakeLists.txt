cmake_minimum_required(VERSION 3.24)
project(ToolscreenNative LANGUAGES C CXX)
option(MACLESS_PORTABLE_DESKTOP "Use shared desktop services" OFF)
option(MACLESS_BUNDLED_FREETYPE "Build FreeType from source" OFF)
if(APPLE AND NOT MACLESS_PORTABLE_DESKTOP)
    enable_language(OBJCXX)
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(APPLE)
    set(CMAKE_OSX_DEPLOYMENT_TARGET "14.0")
endif()
if(MACLESS_JNI_INCLUDE_DIRS)
    set(JNI_INCLUDE_DIRS ${MACLESS_JNI_INCLUDE_DIRS})
else()
    find_package(JNI REQUIRED)
endif()
include(FetchContent)
FetchContent_Declare(imgui
    URL https://codeload.github.com/ocornut/imgui/tar.gz/f5befd2d29e66809cd1110a152e375a7f1981f06
    URL_HASH SHA256=85f4ce357df05bcc331b587f01976f47fb55f19fadf477a7907289686bc3f4c8
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE
)
FetchContent_MakeAvailable(imgui)
add_library(imgui STATIC ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp ${imgui_SOURCE_DIR}/imgui_tables.cpp ${imgui_SOURCE_DIR}/imgui_widgets.cpp ${imgui_SOURCE_DIR}/misc/freetype/imgui_freetype.cpp)
target_include_directories(imgui PUBLIC ${imgui_SOURCE_DIR})
target_compile_definitions(imgui PUBLIC IMGUI_ENABLE_FREETYPE)
if(APPLE AND NOT MACLESS_BUNDLED_FREETYPE)
    find_path(TOOLSCREEN_FREETYPE_INCLUDE ft2build.h PATHS /opt/homebrew/opt/freetype/include PATH_SUFFIXES freetype2 REQUIRED)
    find_library(TOOLSCREEN_FREETYPE_STATIC NAMES libfreetype.a PATHS /opt/homebrew/opt/freetype/lib REQUIRED)
    find_library(TOOLSCREEN_PNG_STATIC NAMES libpng16.a PATHS /opt/homebrew/opt/libpng/lib REQUIRED)
    target_include_directories(imgui PRIVATE ${TOOLSCREEN_FREETYPE_INCLUDE})
    target_link_libraries(imgui PUBLIC ${TOOLSCREEN_FREETYPE_STATIC} ${TOOLSCREEN_PNG_STATIC} bz2 z)
else()
    set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
    foreach(feature ZLIB BZIP2 PNG HARFBUZZ BROTLI)
        set(FT_DISABLE_${feature} ON CACHE BOOL "" FORCE)
    endforeach()
    FetchContent_Declare(freetype
        URL https://codeload.github.com/freetype/freetype/tar.gz/refs/tags/VER-2-14-1
        URL_HASH SHA256=44bd69d1f0750603410cfd4f26f1c5523c5a3a087a2dfa8639b757dc7c6677be
        DOWNLOAD_EXTRACT_TIMESTAMP TRUE
    )
    FetchContent_MakeAvailable(freetype)
    set_target_properties(freetype PROPERTIES POSITION_INDEPENDENT_CODE ON)
    target_link_libraries(imgui PUBLIC freetype)
endif()
set_target_properties(imgui PROPERTIES POSITION_INDEPENDENT_CODE ON)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
set(TOOLSCREEN_FONT_HEADER ${CMAKE_CURRENT_BINARY_DIR}/generated/fonts.hpp)
add_custom_command(OUTPUT ${TOOLSCREEN_FONT_HEADER}
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/embed_fonts.py ${CMAKE_CURRENT_SOURCE_DIR}/native/assets/fonts ${TOOLSCREEN_FONT_HEADER}
    DEPENDS tools/embed_fonts.py native/assets/fonts/Inter-Regular.ttf native/assets/fonts/Inter-Medium.ttf native/assets/fonts/Inter-SemiBold.ttf native/assets/fonts/JetBrainsMono-Regular.ttf native/assets/fonts/Monocraft.ttf
)
add_library(toolscreen SHARED ${TOOLSCREEN_FONT_HEADER} native/core.cpp native/input.cpp native/features.cpp native/overlays.cpp native/bridge.cpp native/ui.cpp native/storage.cpp)
target_include_directories(toolscreen PRIVATE native ${JNI_INCLUDE_DIRS} ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_link_libraries(toolscreen PRIVATE imgui)
if(APPLE AND NOT MACLESS_PORTABLE_DESKTOP)
    target_sources(toolscreen PRIVATE native/platform.mm native/ninja_macos.mm)
    set_source_files_properties(native/platform.mm native/ninja_macos.mm PROPERTIES COMPILE_FLAGS "-fobjc-arc")
    target_link_libraries(toolscreen PRIVATE "-framework AppKit" "-framework UniformTypeIdentifiers" "-framework WebKit" "-framework ScreenCaptureKit" "-framework CoreVideo" "-framework CoreMedia" "-framework CoreGraphics")
    set_target_properties(toolscreen PROPERTIES MACOSX_RPATH ON)
else()
    target_sources(toolscreen PRIVATE native/platform_desktop.cpp native/ninja_desktop.cpp)
    target_compile_definitions(toolscreen PRIVATE MACLESS_PORTABLE_DESKTOP)
    set(json_header ${CMAKE_CURRENT_BINARY_DIR}/generated/nlohmann/json.hpp)
    if(NOT EXISTS ${json_header})
        file(DOWNLOAD https://raw.githubusercontent.com/nlohmann/json/v3.12.0/single_include/nlohmann/json.hpp ${json_header}
            EXPECTED_HASH SHA256=aaf127c04cb31c406e5b04a63f1ae89369fccde6d8fa7cdda1ed4f32dfc5de63 TLS_VERIFY ON)
    endif()
endif()
if(WIN32)
    add_executable(macless-capture native/capture_windows.cpp)
    target_link_libraries(macless-capture PRIVATE d3d11 dxgi runtimeobject ole32 uuid)
    if(MINGW)
        target_link_options(macless-capture PRIVATE -static-libgcc -static-libstdc++ -static)
    endif()
    set_target_properties(toolscreen macless-capture PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR} RUNTIME_OUTPUT_DIRECTORY_RELEASE ${CMAKE_CURRENT_BINARY_DIR})
    set_target_properties(toolscreen PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON PREFIX "")
    if(MINGW)
        target_link_options(toolscreen PRIVATE -static-libgcc -static-libstdc++ -static)
    endif()
endif()
if(MSVC)
    target_compile_options(toolscreen PRIVATE /W4 /WX /permissive- /utf-8)
else()
    target_compile_options(toolscreen PRIVATE -Wall -Wextra -Wpedantic -Werror)
    if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
        target_compile_options(toolscreen PRIVATE -Wno-misleading-indentation)
    endif()
endif()
